@dataclass(frozen=True)
class BillingCaseDetailDTO:
    billing_case: BillingCase

    requirement_assessments: tuple[
        BillingRequirementAssessment,
        ...
    ]

    code_evaluations: tuple[
        BillingCodeEvaluation,
        ...
    ]

    missing_information: tuple[
        MissingBillingInformation,
        ...
    ]

    monitoring_summary: BillingMonitoringSummary | None
    time_summary: BillingTimeSummary | None

    provider_confirmation: (
        ProviderBillingConfirmation | None
    )

    billing_review: BillingReview | None
    claim_preparation: ClaimPreparation | None

    latest_submission: ClaimSubmission | None
    latest_outcome: ClaimOutcome | None

    device_type: str | None
